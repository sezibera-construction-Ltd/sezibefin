# Housekeeping files that carry no search value

# Favicons, called out explicitly. Google will not show an icon beside a
# search result unless it can fetch the file, and a broad Disallow added here
# later would take the icon out of the results with nothing to show that it
# had happened.

Allow: /assets/images/icon-*.png
Allow: /assets/images/favicon.svg
Allow: /assets/images/apple-touch-icon.png
Allow: /site.webmanifest
